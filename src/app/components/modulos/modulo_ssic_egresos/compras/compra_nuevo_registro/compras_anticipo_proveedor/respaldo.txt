import { Component, OnInit } from '@angular/core';

import { subscribe } from 'diagnostics_channel';
import { ValidatorServService } from '../../../../../servicios/validator-serv.service';

@Component({
  selector: 'app_compras_proveedores_anticipo',
  standalone: false,
  
  templateUrl: './proveedores-anticipo.component.html',
  styleUrls: [
    '../../../../../styles/datatable.css',
    '../../../../../styles/input_group.css',
    '../../../../../styles/buttons.css',
    '../../../../../styles/modals.css',
    '../../../../../styles/proveedores.css',

    '../../../../../styles/ubicaciones.css',
    '../../egresos.css',
    './proveedores-anticipo.component.css'
  ]
})
export class ProveedoresAnticipoComponent implements OnInit{
  searchProvList:any;
  lista_proveedores:any = [];
  public proveedorSeleccionado:string = "";
  catalogo_monedas_api:any = [];
  public moneda_codigo:string = "MXN";
  public moneda_decimales:string = "2";
  public moneda_codigo_selected:string = "MXN";
  public moneda_decimales_selected:string = "";
  public tipo_cambio:string = "1.00";
  public cantidad_anticipo:number = 0;

  constructor(
    private provSer:ProveedoresService,
    private provMon:MonedasService,
    private validator:ValidatorServService
  ){
  }

  ngOnInit():void{
    this.lista_Prov();
    this.monedasCatalogoApi();
  }


  lista_Prov(){
    this.provSer.catalogoProveedores_General().subscribe(
      response => {
        if(response.status == 'success'){
          this.lista_proveedores = response.proveedores;
          console.log(this.lista_proveedores);
        }
      },
      error => {
        console.log(error);
      }
    );
  }

  verProveedor(row:any){
    this.proveedorSeleccionado = this.proveedorSeleccionado === row ? null : row; 
  }

  monedasCatalogoApi(){
    this.provMon.getApiMonedasCatalogo().subscribe(
      response => {
        if(response.status == 'success'){
          this.catalogo_monedas_api = response.monedas;
          console.log(this.catalogo_monedas_api);
        }
      }
    )
  }

  valMoned(object:any){
    console.log(object.value)
    if (object.value != '' && this.validator.filtroAlfaNumerico(object.value) == true) {
      for (let val = 0; val < this.catalogo_monedas_api.length ; val++) {
        const esval = this.catalogo_monedas_api[val];
        if (esval['langEN'] == object.value) {
          this.validator.correctoInputRow(object);
          this.moneda_codigo_selected = esval['code'];
          this.moneda_decimales_selected = esval['decimales'];
          return;
        } else {
          this.validator.errorInputRow(object);
          this.moneda_codigo_selected = "";
          this.moneda_decimales_selected = "";
        }
      }
    } else { 
      this.validator.errorInputRow(object);
      this.moneda_codigo_selected = "";
      this.moneda_decimales_selected = "";
    }
  }

  activaTipoCambio(){
    return (this.moneda_codigo_selected == this.moneda_codigo);
  }

  editTipoCambio(event:any){
    if(event.value != '' && this.validator.filtroNum(event.value) == true){
      this.tipo_cambio = event.value;
      this.validator.correctoInputRow(event);
    } else {
      this.tipo_cambio = "1.00";
      this.validator.errorInputRow(event);
    }
  }

  keyupImporteAnticipo(object:any){
    if(object.value != '' && this.validator.filtroNum(object.value) == true){
      this.cantidad_anticipo = object.value;
      this.validator.correctoInputRow(object);
    } else {
      this.cantidad_anticipo = 0;
      this.validator.errorInputRow(object);
    }
  }

  valida_registro(){
    return (this.moneda_codigo_selected == this.moneda_codigo);
  }

}
