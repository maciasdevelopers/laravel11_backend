  async cargaXmlCompra(e: any, objeto: any): Promise<void> {
    const doc_xml = objeto.files[0];

    const xml_tipo = doc_xml.type == 'text/xml' && doc_xml.name.toLowerCase().endsWith('.xml');

        //await this.lecturaInternaXML(objeto);
      } catch (error) {
        this.primeAlerts.add({ severity: 'error', summary: 'SOS-México informa: ', detail: 'Error en el procesamiento de información' });
      }
    } else {
      this.primeAlerts.add({ severity: 'error', summary: 'SOS-México informa: ', detail: !xml_tipo ? 'El archivo no es un XML.' : 'El archivo supera el límite de 2MB.' });

    validacion_pdf ? this.validator.correctoInputRow(objeto) : this.validator.errorInputRow(objeto);

      this.primeAlerts.add({ severity: 'error', summary: 'SOS-México informa: ', detail: mensajeError });
    }
  }

  public droppedLogistica(files: NgxFileDropEntry[]) {
    // 1. Limpiamos estados para recibir el nuevo set de archivos
    this.anexosLogisticaFiles = files;
    this.anexosLogisticaNames = [];
    this.anexosLogisticaDocs = [];

    for (let i = 0; i < files.length; i++) {
      const droppedFile = files[i];

      if (droppedFile.fileEntry.isFile) {
        const fileEntry = droppedFile.fileEntry as FileSystemFileEntry;

        fileEntry.file((file: File) => {
          const typoElement = file.type;
          const nameFile = file.name;
          const sizeFile = file.size;

          // 2. Validación de tipos y tamaño (2MB)
          const allowedTypes = ['application/pdf', 'text/xml', 'image/jpeg', 'image/jpg', 'image/png'];
          const isAllowedType = allowedTypes.includes(typoElement);
          const isAllowedSize = sizeFile <= 2000000;

          if (isAllowedType && isAllowedSize) {
            // 3. Inserción directa y limpia
            // Ya no necesitamos el ciclo for (j...) porque limpiamos al inicio
            this.anexosLogisticaNames.push({ "typoElement": typoElement, "nameFile": nameFile });
            this.anexosLogisticaDocs.push(file);

            console.log(`Archivo aceptado: ${nameFile}`);
            this.cd.detectChanges();
          } else {
            // 4. Manejo de errores específico
            const mensajeError = !isAllowedSize ? `El archivo ${nameFile} excede el tamaño permitido (2MB).` : `El archivo ${nameFile} tiene un formato no permitido (PDF, XML, JPG, PNG).`;

            Swal.fire({
              position: 'top-end',
              icon: 'warning',
              title: mensajeError,
              showConfirmButton: false,
              timer: 3000
            });

            // Opcional: Remover de la lista de visualización si falló la validación técnica
            const index = this.anexosLogisticaFiles.findIndex(f => f.relativePath === droppedFile.relativePath);
            if (index > -1) this.anexosLogisticaFiles.splice(index, 1);
            this.cd.detectChanges();
          }
        });
      }
    }
  }

  get validaFormInicioTransitoCompra():boolean{
    const m = this.logisticBuyModel;

    // 1. Validaciones base obligatorias
    let valido = m.fecha_real_salida !== '' && 
                 m.operador_nombre !== '' && 
                 m.tentativaLlegadaLugarDestino !== '';

    // 2. Validaciones condicionales por tipo de transporte
    if (m.transporte_tipo === 'TERRESTRE') {
      valido = valido && m.transporte_placas !== '';
    } else if (m.transporte_tipo === 'MARITIMO') {
      valido = valido && m.numeroContenedor !== '' && m.bookingNumber !== '';
    } else if (m.transporte_tipo === 'AEREO') {
      valido = valido && m.guiaAerea !== '';
    } else {
      // Si el tipo de transporte está vacío o no coincide, no se aprueba
      valido = false; 
    }

    // 3. Validación condicional por destino
    if (m.tipoLugarDestino === 'otro') {
      valido = valido && m.descripcionLugarDestino !== '';
    }

    // 4. Validar que al menos las partidas tengan sus datos completos si es obligatorio
    const partidasValidas = this.listado_articulos.every((art:any) => art.sku && art.sku !== '');
    
    // Asignamos el estado final al guardián del botón submit
    return valido && partidasValidas;
  }

CREATE TABLE eegr_compras_logistica_bitacora (
  id INT(10) UNSIGNED PRIMARY KEY NOT NULL AUTO_INCREMENT,
  compra_relacionada INT(10) NOT NULL,
  estado_alcanzado VARCHAR(30) NOT NULL COMMENT 'salida_transito, en_transito, aduana, arribo_destino',
  tipo_transporte VARCHAR(20) NOT NULL COMMENT 'terrestre, maritimo, aereo',
  
  -- Fechas de Control Operativo
  fecha_real_salida INT(10) UNSIGNED,
  fecha_tentativa_llegada INT(10) UNSIGNED,
  
  -- Archivos de Control Fiscal (CFDI Traslado / Carta Porte)
  cfdi_xml_url TEXT DEFAULT NULL,
  cfdi_pdf_url TEXT DEFAULT NULL,
  
  -- Operador de la Unidad
  operador_nombre VARCHAR(150) NOT NULL,
  operador_telefono VARCHAR(25) DEFAULT NULL,
  
  -- Campos Polimórficos Dinámicos (Estructurados)
  transporte_identificador_principal VARCHAR(100) DEFAULT NULL COMMENT 'Placas Tractor / No. Contenedor / No. Guía Aérea (AWB)',
  transporte_identificador_secundario VARCHAR(100) DEFAULT NULL COMMENT 'Placas Remolque / Booking Number / Código de Vuelo',
  transporte_permiso_autorizacion VARCHAR(100) DEFAULT NULL COMMENT 'Permiso SCT-SICT / Patente',
  
  -- Destino de la Mercancía
  tipo_lugar_destino VARCHAR(30) NOT NULL COMMENT 'establecimiento_final, otro',
  direccion_destino_especifica TEXT DEFAULT NULL,
  direccion_destino_final int(10) DEFAULT NULL,
  
  -- Notas Generales
  observaciones TEXT DEFAULT NULL,
  
  -- Campos de Auditoría del Sistema Contable
  empresa_vinculada INT(10) NOT NULL,
  usuario_registra INT(10) NOT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  -- Restricciones de Integridad
  FOREIGN KEY (compra_relacionada) REFERENCES eegr_compras (id) ON DELETE CASCADE,
  FOREIGN KEY (empresa_vinculada) REFERENCES main_empresas (id) ON DELETE CASCADE,
  FOREIGN KEY (usuario_registra) REFERENCES teci_usuarios_catalogo (id),
  
  -- Índices para optimizar velocidad de respuesta
  INDEX idx_compra_estado (compra_relacionada, estado_alcanzado),
  INDEX idx_empresa_vinculada (empresa_vinculada)
);


CREATE TABLE eegr_compras_logistica_transportes (
  id INT(10) UNSIGNED PRIMARY KEY NOT NULL AUTO_INCREMENT,
  logistica_bitacora_id INT(10) UNSIGNED NOT NULL,
  tipo_transporte VARCHAR(20) NOT NULL COMMENT 'TERRESTRE, MARITIMO, AEREO',
  
  -- Datos Operativos de la Unidad
  operador_nombre VARCHAR(150) NOT NULL,
  operador_telefono VARCHAR(25) DEFAULT NULL,
  identificador_principal VARCHAR(100) DEFAULT NULL COMMENT 'Placas Tractor / Contenedor / Guía',
  identificador_secundario VARCHAR(100) DEFAULT NULL COMMENT 'Placas Remolque / Booking / Vuelo',
  permiso_autorizacion VARCHAR(100) DEFAULT NULL,
  
  -- 📍 Puntos de Origen y Destino Específicos de este viaje
  direccion_origen TEXT NOT NULL COMMENT 'Dirección de salida física (Bodega Proveedor)',
  tipo_lugar_destino VARCHAR(30) NOT NULL COMMENT 'establecimiento_final, otro',
  direccion_destino_especifica TEXT NOT NULL COMMENT 'Dirección de entrega física',
  
  -- Control Fiscal (Archivos)
  cfdi_xml_url TEXT DEFAULT NULL,
  cfdi_pdf_url TEXT DEFAULT NULL,
  
  FOREIGN KEY (logistica_bitacora_id) REFERENCES eegr_compras_logistica_bitacora (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
