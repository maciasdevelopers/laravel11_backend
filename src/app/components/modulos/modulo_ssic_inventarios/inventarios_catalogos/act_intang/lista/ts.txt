//area sat
checkSatInput(event:any){
  //alert(event.value);
  let catSatHiddenToken = $(event).parent("div").find("input.catSatHiddenToken");
  let h6Satcontent = $(event).parent("div").parent("div").parent("div.data-reg").find("h6.h6SatDescripcion");
  if (event.value.length == 8) {
    this._catSat.catSatClaveInput(event.value).subscribe(
      //console.log(response);
      response => {
        if (response.status == 'success') {
          h6Satcontent.html("Descripción de código de SAT: "+response.catSatDescripcion);
          catSatHiddenToken.val(response.catSatToken);
        }
      },
      error => {
        console.log(error);
      }
    )
  }
}

checkSat(event:any){
  this.filtroBuscaSat = event.value;
  //buscaClaveSat

}

keyupSat(event:any){
  //alert(event.value);
  if (event.value != '' && event.value.length >= 3) {
    if (this.filtroBuscaSat == 'codigo' && event.value.length >= 3 && event.value.length <= 8) {
      this._catSat.catSatProdServClave(event.value).subscribe(
        response => {
          if (response.status == 'success') {
            this.arrayCatSat = response.catSat;
          }
        },
        error => {
          console.log(error);
        }
      )
    }

    if (this.filtroBuscaSat == 'descripcion' && event.value.length >= 3 && event.value.length <= 8) {
      this._catSat.catSatProdServDescripcion(event.value).subscribe(
        response => {
          if (response.status == 'success') {
            this.arrayCatSat = response.catSat;
          }
        },
        error => {
          console.log(error);
        }
      )
    }
  } else {
    this.catSatProdServLimit();
  }
}

              console.log(response.datosActivo);
              this.datosDetalleActivoIntangible = response.datosActivo;
              //datosDetalleActivoIntangible:any = []; public modeloPerfilActIntangible: activoIntangibleAngularModelo;

              this.modeloPerfilActIntangible.clasificacion = response.datosActivo[0]['token_clasificacion_intang'];
              this.modeloPerfilActIntangible.clave_sat = response.datosActivo[0]['clave'];
              this.modeloPerfilActIntangible.descripcion_activo = response.datosActivo[0]['concepto'];
              this.modeloPerfilActIntangible.amortizacionMetodo = response.datosActivo[0]['tipo_amortizacion'];
              this.modeloPerfilActIntangible.amortizacionContable = response.datosActivo[0]['amortizacion_contable'];
              this.modeloPerfilActIntangible.amortizacionFiscal = response.datosActivo[0]['amortizacion_fiscal'];
              for (let a = 0; a < response.datosActivo[0]['uso_cfdi'].length; a++) {
                if (response.datosActivo[0]['uso_cfdi'][a]['selecttkn'] == true) {
                  this.modeloActIntangible.cfdi = response.datosActivo[0]['uso_cfdi'][a]['token_uso_cfdi'];
                  this.modeloPerfilActIntangible.cfdi = response.datosActivo[0]['uso_cfdi'][a]['token_uso_cfdi'];
                }
              }

              if (response.datosActivo[0]['periodo_amortizacion'] == '86400') {
                this.modeloPerfilActIntangible.periodoAmortizacion = 'periodDay';
                this.modeloActIntangible.periodoAmortizacion = 'periodDay';
              } else if (response.datosActivo[0]['periodo_amortizacion'] == '604800') {
                this.modeloPerfilActIntangible.periodoAmortizacion = 'periodWeek';
                this.modeloActIntangible.periodoAmortizacion = 'periodWeek';
              } else if (response.datosActivo[0]['periodo_amortizacion'] == '2629743') {
                this.modeloPerfilActIntangible.periodoAmortizacion = 'periodMonth';
                this.modeloActIntangible.periodoAmortizacion = 'periodMonth';
              } else if (response.datosActivo[0]['periodo_amortizacion'] == '31556926') {
                this.modeloPerfilActIntangible.periodoAmortizacion = 'periodYear';
                this.modeloActIntangible.periodoAmortizacion = 'periodYear';
              }

              for (let i = 0; i < response.datosActivo[0]['proveedores'].length; i++) {
                let arrayPrv = [];
                const proveedoresFor = response.datosActivo[0]['proveedores'][i];
                if (proveedoresFor['asigned_clave'] != '') {
                    arrayPrv[0] = proveedoresFor['token_cat_proveedores'];
                    arrayPrv[1] = proveedoresFor['asigned_clave'];
                    this.arrayClaveProvActF.push(arrayPrv);
                }
              }
              this.modeloPerfilActIntangible.proveedor = this.arrayClaveProvActF;
              console.log(this.arrayClaveProvActF);
              //console.log(desgloseAlm.desgloImportTkn)
              //this.tknSatDetalle
            }
          },
          error => {
            console.log(error);
          }
        )
