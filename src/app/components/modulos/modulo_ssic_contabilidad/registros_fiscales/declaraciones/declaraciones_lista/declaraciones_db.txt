CREATE TABLE cfdi_vinculacion_compras (
  comprobante_fiscal int(10) NOT NULL,
  compra_vinculada int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (compra_vinculada) references eegr_compras (id)
);

CREATE TABLE cfdi_vinculacion_reembolsos (
  comprobante_fiscal int(10) NOT NULL,
  reembolso_vinculado_main int(10) NOT NULL,
  reembolso_vinculado_soli int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (reembolso_vinculado_main) references terc_reembolso_main (id),
  foreign key (reembolso_vinculado_soli) references terc_reembolso_solicitud (id)
);

CREATE TABLE cfdi_vinculacion_isn (
  comprobante_fiscal int(10) NOT NULL,
  isn_vinculado int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (isn_vinculado) references vhum_nominas_impuestos (id)
);

CREATE TABLE cfdi_vinculacion_nomina (
  comprobante_fiscal int(10) NOT NULL,
  nomina_main int(10) NOT NULL,
	nomina_recibo int(10) NOT NULL,
	empleado_referenciado int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (nomina_main) references vhum_nominas_main (id),
  foreign key (nomina_recibo) references vhum_nominas_recibos (id),
  foreign key (empleado_referenciado) references vhum_empleados_catalogo (id)
);

CREATE TABLE cfdi_vinculacion_aport_seg_social_imss (
  comprobante_fiscal int(10) NOT NULL,
  comprobante_tipo varchar(9) NOT NULL,
  aport_seg_social_vinculado int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (aport_seg_social_vinculado) references vhum_aportaciones_seguridad_social_main (id)
);

CREATE TABLE cfdi_vinculacion_declaraciones (
  comprobante_fiscal int(10) NOT NULL,
  declaracion_vinculada int(10) NOT NULL,
  foreign key (comprobante_fiscal) references cfdi_comprobantes_fiscales (id),
  foreign key (declaracion_vinculada) references cont_reg_fisc_declaraciones (id)
);
