    calcularProrrateoPorc(){
      var sumatoria_total_prorrat:number = 0;
      this.listaverticalProrrat.forEach((gasto:any) => gasto.total_prorrateo = 0);

      this.arraySelectedProrrateos.forEach((prtSel:any) => {
        let acumuladoTrTotales:number = 0;
        const total_precio_unitario = parseFloat(prtSel.costo_ajustado) || 0;
        prtSel.detalle.forEach((detSelectos:any,index:any) => {
          var total_detalle:any = parseFloat(detSelectos.total_detalle) || 0;
          var totalCompra:any = parseFloat(detSelectos.totalCompra) || 1;
          var porcentaje:any = (parseFloat(detSelectos.porcentaje_juega) || 0) / 100;
  
          var resultCantProrrateo = (parseFloat(detSelectos.cant_art_prorrateo) || 0) * porcentaje;
          detSelectos.result_porcentaje_contable_juega = resultCantProrrateo.toFixed(this.decimales_monedaOrden);
          detSelectos.result_porcentaje_fiscal_juega = resultCantProrrateo.toFixed(this.decimales_monedaOrden);
          var prorrateoUno = resultCantProrrateo * (total_detalle / totalCompra);
          var prorrateoDos = prorrateoUno / (parseFloat(detSelectos.numero_articulos_prorratea) || 1);
          
          detSelectos.total_prorrateo = prorrateoUno.toFixed(this.decimales_monedaOrden);

          if (prtSel.tipo_art_prorrateo == 'mercancia') {
            acumuladoTrTotales += prorrateoDos;
            detSelectos.desv_art_prorrateo = prorrateoDos.toFixed(this.decimales_monedaOrden);
            this.listaverticalProrrat[index]['total_prorrateo'] += prorrateoDos;
          } else {
            detSelectos.desv_art_prorrateo = 'NA';
            acumuladoTrTotales += resultCantProrrateo;
            this.listaverticalProrrat[index]['total_prorrateo'] += resultCantProrrateo;
          }
        });
        console.log(total_precio_unitario);

        sumatoria_total_prorrat += (total_precio_unitario + acumuladoTrTotales);
        prtSel.trTotales = acumuladoTrTotales.toFixed(this.decimales_monedaOrden);
      });
      console.log(sumatoria_total_prorrat);
      
      this.listaverticalProrrat.forEach((gasto: any) => {
        // Convertimos la suma numérica final al formato de moneda de la empresa
        gasto.total_prorrateo = gasto.total_prorrateo.toFixed(this.decimales_monedaOrden);
      });

      this.prorrateoTotal = sumatoria_total_prorrat;
      console.log(this.arraySelectedProrrateos);
      console.log(this.listaverticalProrrat);
    }

    percentProrrateo(event:any,iPosition:any,token_art_detbuy_prorrateo:any){
      const selectos = this.arraySelectedProrrateos[iPosition];
      var percentFor:any = 0;
      if (event.value != '' && this.validator.filtroCosto(event.value) == true && selectos['token_art_detbuy_prorrateo'] == token_art_detbuy_prorrateo) {
        selectos['trPorcentaje_juega'] = parseFloat(event.value);
        for (let i1 = 0; i1 < selectos['detalle'].length; i1++) {
          const detalleSelectos = selectos['detalle'][i1];
          detalleSelectos['porcentaje_juega'] = parseFloat(event.value).toFixed(this.decimales_monedaOrden);
        }

        for (let a = 0; a < this.arraySelectedProrrateos.length; a++) {
          const detalleSelectos = this.arraySelectedProrrateos[a];
          if (detalleSelectos['trPorcentaje_juega'] != "" && this.validator.filtroCosto(detalleSelectos['trPorcentaje_juega']) == true) {
            percentFor = parseFloat(percentFor) + parseFloat(detalleSelectos['trPorcentaje_juega']);
          }
        }

        this.porcentajeTotal = parseFloat(percentFor).toFixed(this.decimales_monedaOrden);

        if (percentFor < 100) {
          this.validator.correctoInputRow(event);
          this.boolCalcular = false;
        } else if (percentFor == 100){
          this.validator.correctoInputRow(event);
          this.boolCalcular = true;
        } else if (percentFor > 100){
          this.validator.errorInputRow(event);
          this.boolCalcular = false;
        }

      } else {
        this.validator.errorInputRow(event);
        this.boolCalcular = false;
        selectos['trPorcentaje_juega'] = parseFloat('0').toFixed(this.decimales_monedaOrden);
        for (let i1 = 0; i1 < selectos['detalle'].length; i1++) {
          const detalleSelectos = selectos['detalle'][i1];
          detalleSelectos['porcentaje_juega'] = parseFloat('0').toFixed(this.decimales_monedaOrden);
        }
        for (let a = 0; a < this.arraySelectedProrrateos.length; a++) {
          const detalleSelectos = this.arraySelectedProrrateos[a];
          if (detalleSelectos['trPorcentaje_juega'] != "" && this.validator.filtroCosto(detalleSelectos['trPorcentaje_juega']) == true) {
            percentFor = parseFloat(percentFor) + parseFloat(detalleSelectos['trPorcentaje_juega']);
          }
        }
        this.porcentajeTotal = parseFloat(percentFor).toFixed(this.decimales_monedaOrden);
      }
    }

    get validaSaveProrrateo():boolean{
      const validacion_fecha_contabilizacion = this.prorrateo_fecha_contabilizacion != "" && this.validator.filtroFecha(this.prorrateo_fecha_contabilizacion);
      
      const mnd = this.catalogo_monedas_api.find((row: any) => row.code === this.prorrateo_moneda_code);
      const validacion_moneda = this.prorrateo_moneda_code != '' && this.validator.filtroAlfaNumerico(this.prorrateo_moneda_code) == true && typeof mnd !== 'undefined';

      const porcentajeTotalNum = parseFloat(this.porcentajeTotal);
    
      // Condición 1: Debe haber al menos un producto/activo seleccionado
      const tieneSeleccionados = this.arraySelectedProrrateos && this.arraySelectedProrrateos.length > 0;
    
      // Condición 2: Debe haber gastos asignados en la tabla vertical
      const tieneGastos = this.listaverticalProrrat && this.listaverticalProrrat.length > 0;

      // Condición 3: El porcentaje de distribución debe ser exactamente 100
      const esPorcentajeCorrecto = porcentajeTotalNum === 100;

      // Retorna true solo si todas las condiciones de integridad se cumplen
      return validacion_fecha_contabilizacion && validacion_moneda && tieneSeleccionados && tieneGastos && esPorcentajeCorrecto;
    }
